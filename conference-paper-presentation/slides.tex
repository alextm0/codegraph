% Slide 8
\begin{frame}{Global structural ranking drives the empirical gain}
  \vspace{5pt}
  \centering
  \resizebox{\textwidth}{!}{
    \begin{tabular}{lrrrr}
        \toprule
        \textbf{Configuration} & \textbf{R@5} & \textbf{R@10} & \textbf{MRR} & \textbf{Zero-Recall} \\
        \midrule
        BM25-only (Seed-Engineered Lexical Baseline) & 46.7\% & 58.0\% & 0.354 & 126 \\
        One-hop Ego Graph & 46.7\% & 51.0\% & 0.339 & 147 \\
        Weighted PPR & 60.7\% & 68.0\% & 0.440 & 96 \\
        \textbf{Uniform PPR ($k=30$, Main Operating Point)} & \textbf{65.0\%} & \textbf{74.0\%} & \textbf{0.447} & \textbf{78} \\
        Uniform PPR ($k=50$, Sensitivity Setting) & 65.0\% & 75.7\% & 0.447 & 73 \\
        \bottomrule
    \end{tabular}
  }
  
  \vspace{15pt}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Significant Gain}
        \textbf{+16.0 percentage points} over the strongest lexical baseline.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Need for Global Ranking}
        \textbf{One-hop expansion underperforms} because local structural expansion is not enough.
      \end{block}
  \end{columns}
\end{frame}

% Slide 10
\begin{frame}{What this project shows}
  \begin{block}{Scientific Takeaways}
    \begin{itemize}
      \item \textbf{Structure-aware retrieval} works better than text-only retrieval for coding agents.
      \vspace{10pt}
      \item A \textbf{deterministic lexical-to-graph pipeline} achieves strong retrieval quality with low latency.
      \vspace{10pt}
      \item The next bottleneck is \textbf{seed reachability}.
    \end{itemize}
  \end{block}
  
  \vspace{15pt}
  \begin{exampleblock}{Current Boundaries}
    Static-analysis dependence, Python-only scope, file-level evaluation.
  \end{exampleblock}
\end{frame}
