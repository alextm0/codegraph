\begin{frame}{What this project shows}
  \vspace{10pt}
  \Large
  \begin{itemize}
    \item Structure-aware retrieval works better than text-only retrieval for coding agents
    \vspace{15pt}
    \item A deterministic lexical-to-graph pipeline achieves strong retrieval quality with low latency
    \vspace{15pt}
    \item The next bottleneck is seed reachability
  \end{itemize}
