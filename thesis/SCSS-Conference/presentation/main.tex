% Slide 2: Ambiguity of Natural Language
\begin{frame}{The Ambiguity of Natural Language in Agentic Coding}
    \begin{itemize}
        \item \textbf{The Everyday Reality:} Developers increasingly rely on AI agents to navigate large repositories.
        \item \textbf{The Friction Point:} Natural language queries are often abstract (e.g., "fix the permissions logic").
        \item \textbf{The Vocabulary Mismatch:} The prompt rarely matches exact identifiers like \texttt{global\_settings.py}.
        \item \textbf{The Result:} Pure text similarity struggles with this ambiguity. We must leverage structural execution semantics instead.
    \end{itemize}
    \vspace{0.3cm}
    \begin{center}

% Slide 3: Code is a Graph, Not Flat Text
\begin{frame}{Code is a Graph, Not Flat Text}
    \begin{itemize}
        \item \textbf{The Paradigm Shift:} Move away from treating code as a flat sequence of text tokens.
        \item \textbf{The Reality of Code:} Code executes as a deterministic network of dependencies.
        \item \textbf{The Solution:} We use Personalized PageRank to propagate relevance structurally, completely avoiding stochastic LLM queries inside the retrieval loop.
    \end{itemize}

% Slide 4b: 4-Entity Model
\begin{frame}{The 4-Entity Model \& Graph Construction}
    \begin{itemize}
        \item \textbf{Methodological Choice:} Deliberately avoid full AST ingestion. Denser representations dilute retrieval mass.
        \item \textbf{Coarse-Grained Entities:} We extract a strict 4-entity model (Files, Classes, Functions, Methods).
        \item \textbf{Explicit Edges:} Entities are connected by pure syntactical dependencies (\texttt{CONTAINS}, \texttt{IMPORTS}, \texttt{CALLS}, \texttt{INHERITS\_FROM}).
        \item \textbf{Graph Execution:} This precise structure is loaded into a graph database (Neo4j/GDS) for fast algorithmic traversal.
    \end{itemize}
    \vspace{0.3cm}
    \begin{center}
        \textit{(No messy file-proximity heuristics are used---only explicit syntax.)}
    \end{center}
\end{frame}

% Slide 5b: Walkthrough
\begin{frame}{Walkthrough - The Anatomy of a Query}
    \begin{enumerate}
        \item \textbf{Query:} "fix SQL compilation"
        \item \textbf{Seeds:} Keywords extracted and mapped to the \texttt{SQLCompiler} node via BM25.
        \item \textbf{Propagation:} Personalized PageRank diffuses relevance outward across structural edges.
        \item \textbf{IDF Penalty:} Generic hubs (like loggers) are mathematically penalized, keeping relevance within the specific execution domain.
        \item \textbf{Context:} The subgraph collapses into a mathematically rigorous, ranked context payload.
    \end{enumerate}
    \vspace{0.5cm}
    \begin{center}
        \large \textbf{\textcolor{MutedTeal}{Achieved deterministically, in under a second.}}
    \end{center}
\end{frame}

% Slide 8: Reachability
\begin{frame}{The Reachability Bottleneck}
    \begin{itemize}
        \item \textbf{The Core Insight:} If the initial seeds miss the target neighborhood entirely, no ranking mechanism can recover them.
        \item \textbf{The Ceiling:} "Seed reachability" is the true ceiling for agent retrieval.
        \item \textbf{The Taxonomy:} This finding distinguishes failures of graph connectivity (structural) from failures of semantic meaning (lexical).
    \end{itemize}
    \vspace{0.5cm}
    \begin{center}
    \begin{tikzpicture}[scale=0.9, transform shape]
