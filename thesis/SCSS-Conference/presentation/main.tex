% Slide 2: Ambiguity of Natural Language
\begin{frame}{The Ambiguity of Natural Language in Agentic Coding}
    \begin{itemize}
        \item Natural language issue reports are inherently abstract (e.g., "fix permissions logic").
        \item Relevant implementations reside in specifically named files (e.g., \texttt{global\_settings.py}).
        \item Pure text similarity methods frequently struggle to bridge this vocabulary gap.
    \end{itemize}
    \vspace{0.3cm}

\end{frame}

% Slide 3: Code is a Graph, Not Flat Text
\begin{frame}{Code is a Graph, Not Flat Text}
    \begin{itemize}
        \item \textbf{Code is not just text.} A repository is a structured network of files, classes, and methods.
        \item \textbf{Explicit Syntax:} Components are connected by explicit relationships (e.g., imports, calls).
        \item \textbf{The Intuition:} Let relevance travel through these structural relationships, not just through token overlap.
    \end{itemize}
    \vspace{0.3cm}
    \begin{center}

% Slide 4b: 4-Entity Model
\begin{frame}{The 4-Entity Model \& Graph Construction}
    \begin{itemize}
        \item \textbf{Nodes:} Files, Classes, Functions, and Methods.
        \item \textbf{Edges:} \texttt{CONTAINS}, \texttt{IMPORTS}, \texttt{CALLS}, and \texttt{INHERITS\_FROM}.
        \item Avoiding full AST-level density prevents the dilution of the relevance signal.
    \end{itemize}

% Slide 5a: Retrieval Phase
\begin{frame}{System Architecture - Retrieval Phase (Overview)}
    \begin{itemize}
        \item \textbf{No LLMs in the retrieval loop.}
        \item Lexical matching extracts seed nodes from the issue description.
        \item Global ranking is computed purely via Personalized PageRank (PPR).
        \item The retrieval loop remains deterministic, low-latency, and reproducible.
    \end{itemize}
    \vspace{0.8cm}
    \begin{center}
    \begin{tikzpicture}[node distance=3cm, auto, scale=0.9, transform shape]

% Slide 5b: Walkthrough
\begin{frame}{Walkthrough - The Anatomy of a Query}
    \begin{enumerate}
        \item \textbf{Issue Parsing:} Extract keywords from the natural language query.
        \item \textbf{Seed Selection:} Map keywords to initial nodes in the graph (e.g., \texttt{SQLCompiler}).
        \item \textbf{Structural Propagation:} Use Personalized PageRank to diffuse relevance along the edges.
        \item \textbf{Hub Penalization:} Mathematically penalize generic utilities to maintain domain focus.
        \item \textbf{Ranked Output:} Return the structurally most relevant nodes as the final context.
    \end{enumerate}
\end{frame}

% Slide 8: Reachability
\begin{frame}{The Reachability Bottleneck}
    \begin{itemize}
        \item \textbf{Reachability Constraint:} The success of structural retrieval is bounded by the initial seed nodes.
        \item If the seeds reach the relevant structural neighborhood, the ranking succeeds.
        \item If the target is unreachable from the seeds, the file cannot be retrieved.
        \item Failures categorize cleanly into two distinct paradigms:
    \end{itemize}
